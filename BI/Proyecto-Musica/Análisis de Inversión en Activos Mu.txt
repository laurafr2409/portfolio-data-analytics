Análisis de Inversión en Activos Musicales


CREATE TABLE SPOTIFY_DATA (
    TRACK_ID             VARCHAR2(50),
    TRACK_NAME           VARCHAR2(255),
    TRACK_NUMBER         NUMBER,
    TRACK_POPULARITY     NUMBER,
    EXPLICIT             VARCHAR2(5),  -- Para cargar los valores 'TRUE'/'FALSE'
    ARTIST_NAME          VARCHAR2(255),
    ARTIST_POPULARITY    NUMBER,
    ARTIST_FOLLOWERS     NUMBER,
    ARTIST_GENRES        VARCHAR2(500),
    ALBUM_ID             VARCHAR2(50),
    ALBUM_NAME           VARCHAR2(255),
    ALBUM_RELEASE_DATE   VARCHAR2(50), -- CRÍTICO: Se define como texto (VARCHAR2) para aceptar formatos mixtos (YYYY-MM-DD y MM/DD/YYYY) sin error.
    ALBUM_TOTAL_TRACKS   NUMBER,
    ALBUM_TYPE           VARCHAR2(50),
    TRACK_DURATION_MIN   NUMBER
);


DROP TABLE SPOTIFY_DATA_CLEANED CASCADE CONSTRAINTS;

-- ETL (limpieza y filtrado)
CREATE TABLE SPOTIFY_DATA_CLEANED AS
SELECT
    T.TRACK_ID,
    T.TRACK_NAME,
    T.TRACK_POPULARITY,
    T.TRACK_DURATION_MIN,
    T.ALBUM_TYPE,
    T.ARTIST_NAME,
    T.ARTIST_POPULARITY,
    T.ARTIST_FOLLOWERS,
    T.ARTIST_GENRES,

    -- **LIMPIEZA DE FECHA (Conversión a DATE):**
    CASE
        WHEN REGEXP_LIKE(T.ALBUM_RELEASE_DATE, '^\d{4}-\d{2}-\d{2}$')
             THEN TO_DATE(T.ALBUM_RELEASE_DATE, 'YYYY-MM-DD')
        WHEN REGEXP_LIKE(T.ALBUM_RELEASE_DATE, '^\d{1,2}/\d{1,2}/\d{4}$')
             THEN TO_DATE(T.ALBUM_RELEASE_DATE, 'MM/DD/YYYY')
        ELSE NULL
    END AS RELEASE_DATE_CLEAN, -- Columna clave de fecha limpia

    -- **CONVERSIÓN DE FLAG EXPLÍCITO (A 1/0):**
    CASE
        WHEN UPPER(T.EXPLICIT) = 'TRUE' THEN 1
        ELSE 0
    END AS IS_EXPLICIT_FLAG

FROM
    SPOTIFY_DATA T
WHERE
    -- **FILTRADO:** Excluir registros sin valor o con fechas que no pudieron ser parseadas
    T.TRACK_POPULARITY IS NOT NULL
    AND T.ARTIST_FOLLOWERS IS NOT NULL
    AND T.ARTIST_FOLLOWERS > 0
    AND T.TRACK_ID IS NOT NULL
    AND (CASE
        WHEN REGEXP_LIKE(T.ALBUM_RELEASE_DATE, '^\d{4}-\d{2}-\d{2}$') THEN 1
        WHEN REGEXP_LIKE(T.ALBUM_RELEASE_DATE, '^\d{1,2}/\d{1,2}/\d{4}$') THEN 1
        ELSE 0
    END = 1);

--SELECT * FROM SPOTIFY_DATA_CLEANED;
 CREATE TABLE DIM_ARTISTA_GENERO AS
SELECT
    ROWNUM AS ARTIST_SK,
    T.ARTIST_NAME,
    T.ARTIST_POPULARITY,
    T.ARTIST_FOLLOWERS,
    T.ARTIST_GENRES
FROM (
    SELECT DISTINCT ARTIST_NAME, ARTIST_POPULARITY, ARTIST_FOLLOWERS, ARTIST_GENRES
    FROM SPOTIFY_DATA_CLEANED
) T;
ALTER TABLE DIM_ARTISTA_GENERO ADD CONSTRAINT PK_DIM_ARTISTA PRIMARY KEY (ARTIST_SK);

CREATE TABLE DIM_TIEMPO AS
SELECT
    TO_NUMBER(TO_CHAR(RELEASE_DATE_CLEAN, 'YYYYMMDD')) AS DATE_SK,
    RELEASE_DATE_CLEAN AS FULL_DATE,
    EXTRACT(YEAR FROM RELEASE_DATE_CLEAN) AS RELEASE_YEAR,
    TRUNC(SYSDATE - RELEASE_DATE_CLEAN) AS CATALOG_AGE_DAYS,
    CASE
        WHEN EXTRACT(YEAR FROM RELEASE_DATE_CLEAN) >= EXTRACT(YEAR FROM SYSDATE) - 5 THEN 'RECENT'
        ELSE 'CATALOGO'
    END AS RELEASE_FLAG
FROM (
    SELECT DISTINCT RELEASE_DATE_CLEAN FROM SPOTIFY_DATA_CLEANED WHERE RELEASE_DATE_CLEAN IS NOT NULL
) T;
ALTER TABLE DIM_TIEMPO ADD CONSTRAINT PK_DIM_TIEMPO PRIMARY KEY (DATE_SK);

CREATE TABLE DIM_CONTENIDO_RIESGO AS
SELECT
    ROWNUM AS CONTENIDO_SK,
    ALBUM_TYPE,
    IS_EXPLICIT_FLAG,
    ALBUM_TYPE || '_' || CASE WHEN IS_EXPLICIT_FLAG = 1 THEN 'RIESGO_ALTO' ELSE 'RIESGO_BAJO' END AS RIESGO_LABEL
FROM (
    SELECT DISTINCT ALBUM_TYPE, IS_EXPLICIT_FLAG FROM SPOTIFY_DATA_CLEANED
) T;
ALTER TABLE DIM_CONTENIDO_RIESGO ADD CONSTRAINT PK_DIM_CONTENIDO PRIMARY KEY (CONTENIDO_SK);

CREATE TABLE FACT_RENDIMIENTO_PISTA AS
SELECT
    ROWNUM AS TRACK_SK,
    -- Claves Foráneas
    DA.ARTIST_SK,
    DT.DATE_SK,
    DC.CONTENIDO_SK,
    -- Medidas
    T.TRACK_POPULARITY,
    T.TRACK_DURATION_MIN
FROM
    SPOTIFY_DATA_CLEANED T
JOIN
    DIM_ARTISTA_GENERO DA ON T.ARTIST_NAME = DA.ARTIST_NAME
JOIN
    DIM_TIEMPO DT ON T.RELEASE_DATE_CLEAN = DT.FULL_DATE
JOIN
    DIM_CONTENIDO_RIESGO DC ON T.ALBUM_TYPE = DC.ALBUM_TYPE
                            AND T.IS_EXPLICIT_FLAG = DC.IS_EXPLICIT_FLAG;
ALTER TABLE FACT_RENDIMIENTO_PISTA ADD CONSTRAINT PK_FACT_RENDIMIENTO PRIMARY KEY (TRACK_SK);
