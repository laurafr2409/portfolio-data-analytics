-- 1. DIMENSION: DIM_MARCA_MODELO (Posicionamiento)
CREATE TABLE DIM_MARCA_MODELO (
    MODELO_ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY (START WITH 1 INCREMENT BY 1) PRIMARY KEY,
    MARCA VARCHAR2(50) NOT NULL,
    MODELO VARCHAR2(100) NOT NULL,
    CARROCERIA VARCHAR2(50),
    SEGMENTO VARCHAR2(50),
    TRANSMISION VARCHAR2(20),
    CONSTRAINT UQ_MARCA_MODELO UNIQUE (MARCA, MODELO) 
);

-- 2. DIMENSION: DIM_ESPECIFICACIONES_BAT (Tecnología)
CREATE TABLE DIM_ESPECIFICACIONES_BAT (
    BAT_ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY (START WITH 1 INCREMENT BY 1) PRIMARY KEY,
    TIPO_BATERIA VARCHAR2(50) NOT NULL,
    PUERTO_CARGA VARCHAR2(20),
    NUMERO_CELDAS NUMBER,
    CONSTRAINT UQ_BAT_SPECS UNIQUE (TIPO_BATERIA, PUERTO_CARGA, NUMERO_CELDAS)
);

-- 3. DIMENSION: DIM_FISICAS (Diseño)
CREATE TABLE DIM_FISICAS (
    FISICA_ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY (START WITH 1 INCREMENT BY 1) PRIMARY KEY,
    LONGITUD_MM NUMBER,
    ANCHO_MM NUMBER,
    ALTURA_MM NUMBER,
    ASIENTOS NUMBER,
    CONSTRAINT UQ_FISICAS UNIQUE (LONGITUD_MM, ANCHO_MM, ALTURA_MM, ASIENTOS)
);

INSERT INTO DIM_MARCA_MODELO (MARCA, MODELO, CARROCERIA, SEGMENTO, TRANSMISION)
SELECT DISTINCT brand, model, car_body_type, segment, drivetrain
FROM STAGE_EV_SPECS
WHERE brand IS NOT NULL AND model IS NOT NULL;

INSERT INTO DIM_ESPECIFICACIONES_BAT (TIPO_BATERIA, PUERTO_CARGA, NUMERO_CELDAS)
SELECT DISTINCT battery_type, fast_charge_port, number_of_cells
FROM STAGE_EV_SPECS;

INSERT INTO DIM_FISICAS (LONGITUD_MM, ANCHO_MM, ALTURA_MM, ASIENTOS)
SELECT DISTINCT length_mm, width_mm, height_mm, seats
FROM STAGE_EV_SPECS;

COMMIT;

-- CREACIÓN DE LA TABLA DE HECHOS
CREATE TABLE HECHOS_VEHICULOS_EV (
    HECHO_ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY (START WITH 1 INCREMENT BY 1) PRIMARY KEY,
    
    -- CLAVES FORÁNEAS (FK)
    MODELO_ID NUMBER NOT NULL,
    BAT_ID NUMBER NOT NULL,
    FISICA_ID NUMBER NOT NULL,
    
    -- MÉTRICAS
    VELOCIDAD_MAX_KMH NUMBER,
    CAPACIDAD_BAT_KWH NUMBER,
    TORQUE_NM NUMBER,
    EFICIENCIA_WH_KM NUMBER,
    RANGO_KM NUMBER,
    ACELERACION_0_100_S NUMBER,
    CARGA_RAPIDA_KW NUMBER,
    CAPACIDAD_REMOLQUE_KG NUMBER,
    VOLUMEN_CARGA_L NUMBER,
    
    -- Definición de Restricciones de Clave Foránea
    CONSTRAINT FK_HECHOS_MODELO FOREIGN KEY (MODELO_ID) REFERENCES DIM_MARCA_MODELO(MODELO_ID),
    CONSTRAINT FK_HECHOS_BAT FOREIGN KEY (BAT_ID) REFERENCES DIM_ESPECIFICACIONES_BAT(BAT_ID),
    CONSTRAINT FK_HECHOS_FISICA FOREIGN KEY (FISICA_ID) REFERENCES DIM_FISICAS(FISICA_ID)
);

-- CARGA DE LA TABLA DE HECHOS (JOINs para obtener los IDs)
INSERT INTO HECHOS_VEHICULOS_EV (
    MODELO_ID, BAT_ID, FISICA_ID,
    VELOCIDAD_MAX_KMH, CAPACIDAD_BAT_KWH, TORQUE_NM, EFICIENCIA_WH_KM, 
    RANGO_KM, ACELERACION_0_100_S, CARGA_RAPIDA_KW, CAPACIDAD_REMOLQUE_KG, VOLUMEN_CARGA_L
)
SELECT 
    dmm.MODELO_ID, deb.BAT_ID, df.FISICA_ID,
    s.top_speed_kmh, s.battery_capacity_kWh, s.torque_nm, 
    s.efficiency_wh_per_km, s.range_km, s.acceleration_0_100_s, 
    s.fast_charging_power_kw_dc, s.towing_capacity_kg, s.cargo_volume_l
FROM 
    STAGE_EV_SPECS s
JOIN 
    DIM_MARCA_MODELO dmm ON s.brand = dmm.MARCA AND s.model = dmm.MODELO
JOIN 
    DIM_ESPECIFICACIONES_BAT deb ON s.battery_type = deb.TIPO_BATERIA 
        AND s.fast_charge_port = deb.PUERTO_CARGA
        AND s.number_of_cells = deb.NUMERO_CELDAS
JOIN 
    DIM_FISICAS df ON s.length_mm = df.LONGITUD_MM
        AND s.width_mm = df.ANCHO_MM
        AND s.height_mm = df.ALTURA_MM
        AND s.seats = df.ASIENTOS;

COMMIT;

DROP TABLE STAGE_EV_SPECS PURGE;

SELECT
    -- Atributos Descriptivos de Marca y Modelo
    DMM.MARCA AS "Marca",
    DMM.MODELO AS "Modelo",
    DMM.CARROCERIA AS "Tipo_Carroceria",
    DMM.SEGMENTO AS "Segmento_Mercado",
    
    -- Atributos Descriptivos de Batería y Tecnología
    DEB.TIPO_BATERIA AS "Tipo_Bateria",
    DEB.PUERTO_CARGA AS "Puerto_Carga",
    
    -- Atributos Descriptivos Físicos
    DF.LONGITUD_MM AS "Longitud_mm",
    DF.ASIENTOS AS "Num_Asientos",
    
    -- Métricas (Hechos)
    HVE.CAPACIDAD_BAT_KWH AS "Capacidad_Bateria_kWh",
    HVE.RANGO_KM AS "Rango_WLTP_km",
    HVE.ACELERACION_0_100_S AS "Aceleracion_0_100_s",
    HVE.CARGA_RAPIDA_KW AS "Carga_Rapida_kW",
    HVE.VELOCIDAD_MAX_KMH AS "Velocidad_Max_kmh",
    HVE.EFICIENCIA_WH_KM AS "Eficiencia_Wh_km"

FROM 
    -- 1. Tabla de Hechos (Central)
    HECHOS_VEHICULOS_EV HVE

-- 2. Unión a la Dimensión Marca/Modelo
JOIN 
    DIM_MARCA_MODELO DMM ON HVE.MODELO_ID = DMM.MODELO_ID

-- 3. Unión a la Dimensión Especificaciones Batería
JOIN 
    DIM_ESPECIFICACIONES_BAT DEB ON HVE.BAT_ID = DEB.BAT_ID

-- 4. Unión a la Dimensión Físicas
JOIN 
    DIM_FISICAS DF ON HVE.FISICA_ID = DF.FISICA_ID;
